class Solution {
    public int maximumSaleItems(int[][] items, int budget) {
        Arrays.sort(items, (a, b) -> Integer.compare(a[1], b[1]));
        int max = 0;
        for(int[] item : items) {
            max = Math.max(max, item[0]);
        }
        int[] freq = new int[max + 1];
        for(int[] item : items) {
            freq[item[0]]++;
        }
        for(int i=1; i<=max; i++) {
            if(freq[i] > 0) {
                for(int j=i*2; j<=max; j+=i) {
                    freq[i] += freq[j];
                }
            }
        }

        int min = items[0][1];
        int result = 0;
        for(int[] item : items) {
            int factor = item[0], price = item[1];
            if(price - min > min) {
                break;
            }
            int count = freq[factor] - 1;
            int take = Math.min(count, budget / price);
            result += take * 2;
            budget -= take * price;
        }
        result += budget / min;

        return result;
    }
}
